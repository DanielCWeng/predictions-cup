def require(condition: bool, reason: str) -> None:
    if not condition:
        raise RuntimeError(reason)


def git_file(repo: Path, commit: str, path: str) -> bytes:
    return subprocess.check_output(
        ["git", "-C", str(repo), "show", f"{commit}:{path}"], timeout=20
    )


def git_blob_sha(payload: bytes) -> str:
    return hashlib.sha1(f"blob {len(payload)}\0".encode() + payload).hexdigest()


def extract_exact(repo: Path, destination: Path, ref: str) -> None:
    for file in FILES:
        contents = git_file(repo, ref, file)
        if file == ARCHIVE_PATH:
            require(git_blob_sha(contents) == BLOBS[ref], "EXACT_BLOB_SHA_MISMATCH")
        target = destination / file.removeprefix("src/")
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_bytes(contents)
    for package in (destination / "predictions_cup", destination / "predictions_cup" / "research"):
        (package / "__init__.py").write_text("", encoding="utf-8")


def sig_event(ms: int, seq: int, tournament: str, *, stale_native: bool = False) -> dict:
    receive = BASE + timedelta(milliseconds=ms)
    publisher = receive - timedelta(milliseconds=100 if not stale_native else 10_000)
    frame = {"delivery": {"revision": seq, "previousRevision": seq - 1,
            "sourceSequenceFrom": seq, "sourceSequenceThrough": seq},
             "books": [{"exchangeId": 912,
             "asOf": {"sequence": seq, "at": publisher.isoformat()},
             "bids": [{"price": "0.40", "quantity": "2.9"},
                      {"price": "0.39", "quantity": "20.5"}],
             "asks": [{"price": "0.45", "quantity": "2.9"},
                      {"price": "0.46", "quantity": "20.5"}]}]}
    return {"raw_json": json.dumps(frame, separators=(",", ":")),
            "local_receive_at": receive,
            "topic": f"tournament:{tournament}:market:223",
            "connection_epoch": 1, "session_id": "offline-synthetic-1"}


def pm_event(ms: int, token: str, *, source_lag_to_sig: int = 3000,
             bad_future: bool = False, bid: str = "0.61") -> dict:
    recorded = BASE + timedelta(milliseconds=ms)
    # SIG received at minute +4000ms. source_lag_to_sig=5000/5001 probe.
    publisher = recorded + timedelta(milliseconds=3000-source_lag_to_sig)
    if bad_future:
        publisher = recorded + timedelta(seconds=1)
    return {"token_id": token, "recorded_at": recorded,
            "source_timestamp": publisher, "state_observed_at": publisher,
            "bids": [{"price": bid, "size": "3.5"},
                     {"price": "0.59" if bid == "0.61" else "0.35", "size": "20.2"}],
            "asks": [{"price": "0.65", "size": "3.5"},
                     {"price": "0.66", "size": "20.2"}]}


def generate_fixture(root: Path, package_root: Path) -> None:
    import pyarrow as pa
    import pyarrow.parquet as pq
    sys.path.insert(0, str(package_root))
    from predictions_cup.research.native_paired_source_bridge import PM_EXACT_SAME_YES, SIG_TOURNAMENT
    token = PM_EXACT_SAME_YES["223"][2]
    wrong_token = PM_EXACT_SAME_YES["224"][2]
    for minute in range(120):
        base_ms = minute * 60_000
        ts = BASE + timedelta(minutes=minute)
        raw = [sig_event(base_ms+4000, minute+1, SIG_TOURNAMENT,
                         stale_native=(minute == 36))]
        for j in range(48):
            raw.append({"raw_json": '{"books":[{"exchangeId":999999}]}',
                        "local_receive_at": ts + timedelta(milliseconds=4100+j),
                        "topic": "synthetic-unrelated", "connection_epoch": 1,
                        "session_id": "synthetic-unrelated"})
        lag = 5000 if minute == 31 else 5001 if minute == 32 else 3000
        pm = [pm_event(base_ms+1000, token, source_lag_to_sig=lag,
                       bad_future=(minute == 33),
                       bid="0.42" if minute == 34 else "0.61")]
        if minute in (30, 31, 40):
            pm.append(pm_event(base_ms+1000, wrong_token))
        if minute == 35:
            # Nearest PM at +3.5s has stale publisher: old +1s cannot rescue it.
            latest = pm_event(base_ms+3500, token)
            latest["source_timestamp"] = ts - timedelta(seconds=20)
            pm.append(latest)
        for src, stream, rows in (("sig", "raw_events", raw),
                                  ("pm", "depth_snapshots", pm)):
            folder = root/src/stream/ts.strftime("%Y/%m/%d/%H")
            folder.mkdir(parents=True, exist_ok=True)
            shard = folder/(ts.strftime("%Y%m%dT%H%M%SZ") + ".parquet")
            pq.write_table(pa.Table.from_pylist(rows), shard, compression="zstd")

def run(root: Path, source: Path, mode: str) -> dict:
    job = subprocess.run([sys.executable, "-c", WORKER, str(root), str(source), mode],
                         capture_output=True, text=True, timeout=50,
                         env={**os.environ, "PYTHONNOUSERSITE": "1"})
    if job.returncode:
        raise RuntimeError(f"BENCH_CHILD_FAILED mode={mode} returncode={job.returncode} stderr={job.stderr[-2000:]}")
    return json.loads(job.stdout)

        raise RuntimeError("BLOCKER: PYARROW_REQUIRED_NO_PSEUDO_PARQUET") from exc
    with tempfile.TemporaryDirectory(prefix="perf220-offwest-") as directory:
        scratch = Path(directory)
        oldsrc, newsrc = scratch/"old", scratch/"new"
        extract_exact(repo, oldsrc, OLD)
        extract_exact(repo, newsrc, NEW)
        fixture = scratch/"synthetic_archive"
        generate_fixture(fixture, newsrc)
        print("SOURCE_SHA_VERIFIED", json.dumps(BLOBS))
        observations = []
        for round_id in range(4):
            for ref, source, mode in ((OLD,oldsrc,"full_slice"),
                                      (NEW,newsrc,"full_slice"),
                                      (NEW,newsrc,"pilot"),
                                      (OLD,oldsrc,"full_2h")) if round_id%2==0 else (
                                      (NEW,newsrc,"pilot"),
                                      (NEW,newsrc,"full_slice"),
                                      (OLD,oldsrc,"full_slice"),
                                      (OLD,oldsrc,"full_2h")):
                measure = run(fixture, source, mode)
                measure["source_sha"] = ref
                measure["round"] = round_id
                observations.append(measure)
        # All full versus pilot 10m results compare exact arithmetic, not just scalar count.
        canon = [x["contract"] for x in observations if x["mode"] != "full_2h"]
        require(all(x == canon[0] for x in canon), "FAIL_NUMERICAL_OR_PLACEBO_PARITY")
        pilots = [x for x in observations if x["mode"] == "pilot"]
        for row in pilots:
            counts = row["counts"]
            require(row["shards_opened"]["raw_events"] <= 14 and
                    row["shards_opened"]["depth_snapshots"] <= 14,
                    "PILOT_SHARD_CAP_VIOLATED")

                    counts["depth_snapshots_scanned"] <= 35_000,

                    "PILOT_BOTH_STREAM_BYTES_CAP_VIOLATED")

        for ref, mode in ((OLD,"full_slice"),(NEW,"full_slice"),

            entries = [x for x in observations if x["source_sha"]==ref and x["mode"]==mode]

                "cpu_minmax_s": [round(min(x["cpu_s"] for x in entries),6),

                "wall_median_s": round(statistics.median(x["wall_s"] for x in entries),6),
                "peak_rss_max_kib": max(x["peak_rss_kib"] for x in entries),
                "sig_shards": entries[0]["shards_opened"]["raw_events"],
                "pm_shards": entries[0]["shards_opened"]["depth_snapshots"],
                "sig_scanned": entries[0]["counts"]["raw_events_scanned"],
                "pm_scanned": entries[0]["counts"]["depth_snapshots_scanned"],

                                             max(1,entries[0]["counts"]["raw_events_scanned"]+
                                                     entries[0]["counts"]["depth_snapshots_scanned"]),6),
