def sha(p:Path):
    h=hashlib.sha256()
    with p.open("rb") as f:
        for c in iter(lambda:f.read(1<<20),b""): h.update(c)
    return h.hexdigest()

def baseline_rows(*,token,quote,fills,grid_times,window_seconds,own_lookback_seconds,freshness_ns,horizon_seconds):
    own_move,own_valid=ha._feature_move(quote,grid_times,own_lookback_seconds,freshness_ns)
    _,entry_bid,entry_ask,quote_valid=quote.sample(grid_times,freshness_ns=freshness_ns)
    y,_,_,future_bid,future_ask=ha._future_target(quote,grid_times,horizon_seconds,freshness_ns)
    spread=entry_ask-entry_bid
    count=np.zeros(len(grid_times),dtype=float)
    value=np.zeros(len(grid_times),dtype=float)
    unique_participants=np.zeros(len(grid_times),dtype=float)
    window_ns=ha._seconds_ns(window_seconds); second_ns=ha._seconds_ns(1)
    for i,t in enumerate(grid_times):
        cutoff=(t//second_ns)*second_ns
        lo=np.searchsorted(fills.times_ns,cutoff-window_ns,side="left")
        hi=np.searchsorted(fills.times_ns,cutoff,side="left")
        if hi<=lo: continue
        count[i]=hi-lo
        vals=fills.value_usd[lo:hi]
        value[i]=float(np.sum(vals))
        people=set(str(x) for x in fills.maker[lo:hi])|set(str(x) for x in fills.taker[lo:hi])
        unique_participants[i]=len(people)
    baseline=np.column_stack([count,value,unique_participants,own_move,spread])
    valid=own_valid & quote_valid & np.isfinite(y) & np.all(np.isfinite(baseline),axis=1)
    return valid,baseline,y,entry_bid,entry_ask,future_bid,future_ask,own_move

def fit_scores(*,quote_by_token,fills_by_token,train_start_ns,train_end_ns,score_horizon_seconds,score_lookback_seconds,freshness_ns,minimum_history):
    values=defaultdict(list)
    one_second=ha._seconds_ns(1); horizon_ns=ha._seconds_ns(score_horizon_seconds)
    for token,fills in fills_by_token.items():
        quote=quote_by_token.get(token)
        if quote is None or not len(fills.times_ns): continue
        selected=(fills.times_ns>=train_start_ns)&(fills.times_ns+one_second+horizon_ns<train_end_ns)
        indices=np.flatnonzero(selected)
        if not len(indices): continue
        decision=((fills.times_ns[indices]//one_second)+1)*one_second
        own_move,own_valid=ha._feature_move(quote,decision,score_lookback_seconds,freshness_ns)
        future_y,_,_,_,_=ha._future_target(quote,decision,score_horizon_seconds,freshness_ns)
        valid=own_valid & np.isfinite(future_y) & (np.abs(own_move)>1e-12)
        for local,fill_index in enumerate(indices):
            if not valid[local]: continue
            continuation=float(np.sign(own_move[local])*future_y[local])
            for address in {str(fills.maker[fill_index]),str(fills.taker[fill_index])}:
                values[address].append(continuation)
    return {address:float(np.mean(samples)) for address,samples in values.items() if len(samples)>=minimum_history}

def feature(*,fills,times_ns,own_move,scores,window_seconds):
    result=np.zeros(len(times_ns),dtype=float)
    window_ns=ha._seconds_ns(window_seconds); one_second=ha._seconds_ns(1)
    for i,t in enumerate(times_ns):
        cutoff=(t//one_second)*one_second
        lo=np.searchsorted(fills.times_ns,cutoff-window_ns,side="left")
        hi=np.searchsorted(fills.times_ns,cutoff,side="left")
        if hi<=lo or not np.isfinite(own_move[i]) or abs(own_move[i])<=1e-12: continue
        weighted=0.0; total=0.0
        for j in range(lo,hi):
            people={str(fills.maker[j]),str(fills.taker[j])}
            identity_score=float(np.mean([scores.get(p,0.0) for p in people]))
            v=float(fills.value_usd[j]); weighted+=v*identity_score; total+=v
        if total>0: result[i]=float(np.sign(own_move[i])*weighted/total)
    return result

def window(x):
    from datetime import datetime
    return datetime.fromisoformat(x[0]),datetime.fromisoformat(x[1])

def universe(reg):
    rels=[r for r in relationships if r.regime_id==reg]
    low={x["token_id"] for x in prereg["families"]["lowrank"]["lowrank_universe"][reg]}
    all_t={r.target_token_id for r in rels}|{r.reference_token_id for r in rels}|low
    p=tuple(sorted(r["token_id"] for r in identity if r["regime_id"]==reg and r["outcome"]=="Yes" and r["token_id"] in all_t and int(r["fill_rows"] or 0)>=50))
    return p,tuple(sorted(all_t))

def shuffled(fills_by_token,draw,reg):
    out={}
    for token,fs in sorted(fills_by_token.items()):
        seed=int.from_bytes(hashlib.sha256(f"{COMMIT}|role-agnostic|{reg}|{token}|{draw}".encode()).digest()[:8],"big")
        rng=np.random.default_rng(seed)
        slots=np.concatenate([fs.maker.copy(),fs.taker.copy()])
        rng.shuffle(slots)
        n=len(fs.times_ns)
        out[token]=ha.FillSeries(fs.times_ns,fs.value_usd,slots[:n],slots[n:],fs.exchange_taker)
    return out

def run_all(run_id,draw=None,min_history=None):
    evidence=[]
    for reg,(start,end,quotes,fills) in cache.items():
        vf=fills if draw is None else shuffled(fills,draw,reg)
        res=ha.run_participant_oos(
            regime_id=reg,event_family_id=prereg["event_families"][reg],
            quote_by_token=quotes,fills_by_token=vf,start=start,end=end,grid_seconds=30,
            horizon_seconds=H,activity_window_seconds=int(cfg["activity_window_seconds"]),
            own_lookback_seconds=int(cfg["participant_score_lookback_seconds"]),
            score_horizon_seconds=int(cfg["participant_score_horizon_seconds"]),
            score_lookback_seconds=int(cfg["participant_score_lookback_seconds"]),
            minimum_history=int(min_history or cfg["minimum_train_fill_history"]),
            freshness_seconds=fresh,validation=prereg["validation"],run_id=run_id,
        )
        evidence.append(ha.compact_oos(res,session))
    return evidence
