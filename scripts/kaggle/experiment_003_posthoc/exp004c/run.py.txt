def sha(p:Path):
    h=hashlib.sha256()
    with p.open("rb") as f:
        for c in iter(lambda:f.read(1<<20),b""): h.update(c)
    return h.hexdigest()

def one(name):
    m=sorted(INPUT.rglob(name))
    if len(m)!=1: raise SystemExit(f"expected one {name}, got {m}")
    return m[0]

wheel=one("predictions_cup-0.1.0-py3-none-any.whl")
commit=one("COMMIT.txt")
prereg_p=one("preregistration.json")
rel_p=one("relationship_inventory.csv")

from predictions_cup.learning.historical_alpha import (
    FillSeries, compact_oos, load_fill_series, load_identity_rows,
    load_quote_series, load_relationships, run_participant_oos, summarize_evidence,

prereg=json.loads(prereg_p.read_text())
relationships=load_relationships(rel_p,REL_SHA)

identity=load_identity_rows(corpus_root)

session=int(prereg["statistics"]["block_test"]["session_seconds"])

fresh=int(prereg["target"]["quote_freshness_seconds"])

def window(x):
    from datetime import datetime
    return datetime.fromisoformat(x[0]),datetime.fromisoformat(x[1])

def universe(reg):
    rels=[r for r in relationships if r.regime_id==reg]
    low={x["token_id"] for x in prereg["families"]["lowrank"]["lowrank_universe"][reg]}
    all_t={r.target_token_id for r in rels}|{r.reference_token_id for r in rels}|low
    p=tuple(sorted(
        r["token_id"] for r in identity
        if r["regime_id"]==reg and r["outcome"]=="Yes"
        and r["token_id"] in all_t and int(r["fill_rows"] or 0)>=50
    ))
    return p,tuple(sorted(all_t))

cache={}
for reg,w in prereg["regimes"].items():
    p_tokens,all_tokens=universe(reg)
    start,end=window(w)
    cache[reg]=(
        start,end,
        load_quote_series(corpus_root,reg,all_tokens,epsilon=float(prereg["target"]["clamp_epsilon"])),
        load_fill_series(corpus_root,reg,p_tokens),
    )

def shuffled_fills(fills_by_token,draw,reg):
    out={}
    for token,fs in sorted(fills_by_token.items()):
        seed=int.from_bytes(hashlib.sha256(f"{COMMIT}|{reg}|{token}|{draw}".encode()).digest()[:8],"big")
        rng=np.random.default_rng(seed)
        maker=fs.maker.copy(); taker=fs.taker.copy()
        rng.shuffle(maker); rng.shuffle(taker)
        out[token]=FillSeries(fs.times_ns,fs.value_usd,maker,taker,fs.exchange_taker)
    return out

def run_all(run_id,draw=None):
    ev=[]
    for reg,(start,end,quotes,fills) in cache.items():
        vf=fills if draw is None else shuffled_fills(fills,draw,reg)
        res=run_participant_oos(
            regime_id=reg,
            event_family_id=prereg["event_families"][reg],
            quote_by_token=quotes,
            fills_by_token=vf,
            start=start,end=end,
            grid_seconds=30,
            horizon_seconds=H,
            activity_window_seconds=int(cfg["activity_window_seconds"]),
            own_lookback_seconds=int(cfg["participant_score_lookback_seconds"]),
            score_horizon_seconds=int(cfg["participant_score_horizon_seconds"]),
            score_lookback_seconds=int(cfg["participant_score_lookback_seconds"]),
            minimum_history=int(cfg["minimum_train_fill_history"]),
            freshness_seconds=fresh,
            validation=prereg["validation"],
            run_id=run_id,
        )
        ev.append(compact_oos(res,session))
    return summarize_evidence(ev,run_id=run_id,component=run_id,statistics_spec=prereg["statistics"])
