class CsvSink:
    def __init__(self, path: Path, fields: list[str]) -> None:
        self.path = path
        self.fields = fields
        self.handle = path.open("w", newline="", encoding="utf-8")
        self.writer = csv.DictWriter(self.handle, fieldnames=fields, extrasaction="ignore")
        self.writer.writeheader()
        self.rows_written = 0

    def write(self, rows: Iterable[dict[str, Any]]) -> None:
        for row in rows:
            self.writer.writerow({key: row.get(key) for key in self.fields})
            self.rows_written += 1

    def close(self) -> None:
        self.handle.close()


class ParquetSink:
    def __init__(self, path: Path, schema: pa.Schema) -> None:
        self.path = path
        self.schema = schema
        self.writer: pq.ParquetWriter | None = None
        self.rows_written = 0

    def write(self, rows: list[dict[str, Any]]) -> None:
        if not rows:
            return
        table = pa.Table.from_pylist(rows, schema=self.schema)
        if self.writer is None:
            self.writer = pq.ParquetWriter(self.path, self.schema, compression="zstd")
        self.writer.write_table(table)
        self.rows_written += len(rows)

    def close(self) -> None:
        if self.writer is None:
            pq.write_table(pa.Table.from_pylist([], schema=self.schema), self.path)
        else:
            self.writer.close()


class CanonicalTokenSink:
    def __init__(self, root: Path) -> None:
        self.root = root
        self.root.mkdir(parents=True, exist_ok=True)
        self.buffers: dict[str, list[dict[str, Any]]] = defaultdict(list)
        self.writers: dict[str, pq.ParquetWriter] = {}
        self.paths: dict[str, Path] = {}
        self.counts: Counter[str] = Counter()

    def add(self, token: str, row: dict[str, Any]) -> None:
        buf = self.buffers[token]
        buf.append(row)
        self.counts[str(row.get("row_kind"))] += 1
        if len(buf) >= CANON_BUFFER_ROWS:
            self.flush(token)

    def flush(self, token: str) -> None:
        rows = self.buffers.get(token)
        if not rows:
            return
        path = self.paths.setdefault(token, self.root / f"{token}.parquet")
        writer = self.writers.get(token)
        if writer is None:
            writer = pq.ParquetWriter(path, CANON_SCHEMA, compression="zstd")
            self.writers[token] = writer
        writer.write_table(pa.Table.from_pylist(rows, schema=CANON_SCHEMA))
        rows.clear()

    def close(self) -> None:
        for token in list(self.buffers):
            self.flush(token)
        for writer in self.writers.values():
            writer.close()


def load_json(path: Path) -> dict[str, Any]:
    payload = json.loads(path.read_text(encoding="utf-8"))
    if not isinstance(payload, dict):
        raise RuntimeError(f"{path} must contain an object")
    return payload


def write_json(path: Path, value: Any) -> None:
    path.write_text(json.dumps(value, indent=2, sort_keys=True, default=str) + "\n", encoding="utf-8")

def token_string(value: Any) -> str:
    if isinstance(value, memoryview):
        value = value.tobytes()
    if isinstance(value, (bytes, bytearray)):
        return str(int.from_bytes(bytes(value), "big"))
    return str(value)


def token_values_for_type(tokens: set[str], typ: pa.DataType) -> pa.Array:
    ordered = sorted(tokens)
    if pa.types.is_string(typ) or pa.types.is_large_string(typ):
        return pa.array(ordered, type=typ)
    if pa.types.is_binary(typ) or pa.types.is_large_binary(typ) or pa.types.is_fixed_size_binary(typ):
        raw = [int(token).to_bytes(32, "big") for token in ordered]
        return pa.array(raw, type=typ)
    raise RuntimeError(f"unsupported asset_id type: {typ}")


def valid_bbo(bid: Any, ask: Any) -> tuple[float, float] | None:
    try:
        b = float(bid)
        a = float(ask)
    except (TypeError, ValueError):
        return None
    if not (math.isfinite(b) and math.isfinite(a) and 0.0 < b <= a < 1.0):
        return None
    return b, a


def hour_from_path(path: Path) -> pd.Timestamp:
    date = next((part.split("=", 1)[1] for part in path.parts if part.startswith("date=")), None)
    hour = next((part.split("=", 1)[1] for part in path.parts if part.startswith("hour=")), None)
    if date is None or hour is None:
        raise RuntimeError(f"cannot parse hour from {path}")
    return pd.Timestamp(f"{date}T{int(hour):02d}:00:00Z")


def build_universe(
    mapping: dict[str, Any],
    targets: dict[str, Any],
) -> tuple[dict[str, dict[str, Any]], dict[str, str], dict[str, list[str]]]:
    condition_to_tokens = {
        str(condition): [str(token) for token in tokens]
        for condition, tokens in dict(targets["condition_to_tokens"]).items()
    }
    execution: dict[str, dict[str, Any]] = {}
    complement: dict[str, str] = {}
    errors: list[str] = []
    for row in mapping.get("records", []):
        if row.get("mapping_class") not in {PRIMARY_CLASS, SENSITIVITY_CLASS}:
            continue
        direct = row.get("direct_polymarket")
        if not isinstance(direct, dict):
            continue
        token = str(direct.get("mapped_token_id") or "")
        condition = str(direct.get("condition_id") or "")
        pair = condition_to_tokens.get(condition, [])
        if not token or len(pair) != 2 or token not in pair:
            errors.append(f"direct mapping does not bind to binary target pair: {row.get('sig_market_id')}")
            continue
        other = pair[0] if pair[1] == token else pair[1]
        execution[token] = {
            "mapping_class": str(row["mapping_class"]),
            "sig_market_id": str(row.get("sig_market_id") or ""),
            "sig_exchange_id": str(row.get("sig_exchange_id") or ""),
            "condition_id": condition,
            "complement_token": other,
        }
        complement[token] = other
    if errors:
        raise RuntimeError("; ".join(errors[:10]))
    if not execution:
        raise RuntimeError("no direct EXACT/NEAR execution tokens")
    if len(execution) != len(set(execution)):
        raise RuntimeError("duplicate execution token")
    return execution, complement, condition_to_tokens


def dataset_audit(
    root: Path,
    binding: DatasetBinding,
    execution: dict[str, dict[str, Any]],
    targets: dict[str, Any],
) -> dict[str, Any]:
    base = inspect_input(root, binding)
    parquet_files = sorted(root.rglob("*.parquet"))
    schema_counts: Counter[str] = Counter()
    source_rows: Counter[str] = Counter()
    for path in parquet_files:
        pf = pq.ParquetFile(path)
        names = set(pf.schema_arrow.names)
        if {"update_type", "data"}.issubset(names):
            family = "V1_NESTED"
        elif {"event_type", "asset_id", "timestamp"}.issubset(names):
            family = "V3_FLAT" if "sequence" in names else "V2_FLAT"
        else:
            family = "UNKNOWN"
        schema_counts[family] += 1
        source_rows[family] += int(pf.metadata.num_rows)
    exact = sum(1 for meta in execution.values() if meta["mapping_class"] == PRIMARY_CLASS)
    near = sum(1 for meta in execution.values() if meta["mapping_class"] == SENSITIVITY_CLASS)
    result = base.to_dict()
    result["streaming_adapter"] = {
        "adapter_version": "pmxt-v1-v2-v3-stream-v1",
        "target_conditions": len(targets.get("condition_ids", [])),
        "target_tokens": len(targets.get("token_ids", [])),
        "direct_execution_tokens": len(execution),
        "exact_execution_tokens": exact,
        "near_sensitivity_tokens": near,
        "schema_file_counts": dict(schema_counts),
        "schema_row_counts": dict(source_rows),
        "scientific_fill_sources": ["V2_FLAT", "V3_FLAT"],
        "excluded_from_fill_claims": {
            "V1_NESTED": "audited representative V1 files expose book_snapshot/price_change but no last_trade_price event evidence",
        },
        "external_fv_provider": "binary complement as-of: 1 - opposite-token midpoint",
        "external_fv_independent_of_execution_token_mid": True,
        "external_fv_same_venue": True,
        "historical_pm_to_sig_latency_claim": False,
        "trade_event": "last_trade_price only",
        "price_change_is_trade": False,
        "queue_ahead_observed": False,
        "fee_assumption_bound": False,
        "unwind_cost_assumption_bound": False,
        "edge_grid_reference_latency_ms": EDGE_REFERENCE_LATENCY_MS,
    }
    result["passed"] = bool(base.passed and schema_counts["UNKNOWN"] == 0 and len(execution) >= 140)
    return result

def filtered_flat_frame(path: Path, required_tokens: set[str]) -> pd.DataFrame:
    pf = pq.ParquetFile(path)
    names = pf.schema_arrow.names
    if "event_type" not in names or "asset_id" not in names or "timestamp" not in names:
        return pd.DataFrame()
    wanted = [
        name
        for name in (
            "event_type",
            "timestamp",
            "sequence",
            "asset_id",
            "best_bid",
            "best_ask",
            "price",
            "size",
            "side",
            "source_version",
            "fee_rate_bps",
            "transaction_hash",
        )
        if name in names
    ]
    asset_type = pf.schema_arrow.field("asset_id").type
    token_values = token_values_for_type(required_tokens, asset_type)
    parts: list[pd.DataFrame] = []
    for batch in pf.iter_batches(batch_size=BATCH_ROWS, columns=wanted, use_threads=True):
        table = pa.Table.from_batches([batch])
        asset = table["asset_id"].combine_chunks()
        event = table["event_type"].combine_chunks()
        mask = pc.and_(
            pc.is_in(asset, value_set=token_values),
            pc.is_in(event, value_set=pa.array(TARGET_EVENT_TYPES, type=event.type)),
        )
        filtered = table.filter(mask)
        if filtered.num_rows:
            parts.append(filtered.to_pandas())
    if not parts:
        return pd.DataFrame()
    frame = pd.concat(parts, ignore_index=True, sort=False)
    frame["__token"] = frame["asset_id"].map(token_string)
    frame = frame[frame["__token"].isin(required_tokens)].copy()
    if frame.empty:
        return frame
    frame["__ts_ns"] = pd.DatetimeIndex(pd.to_datetime(frame["timestamp"], utc=True)).as_unit("ns").asi8
    frame["__order"] = np.arange(len(frame), dtype=np.int64)
    sort_cols = ["__ts_ns"]
    if "sequence" in frame.columns:
        numeric_seq = pd.to_numeric(frame["sequence"], errors="coerce")
        frame["__sequence"] = numeric_seq.fillna(frame["__order"]).astype(np.int64)
        sort_cols.append("__sequence")
    sort_cols.append("__order")
    frame.sort_values(sort_cols, inplace=True, kind="stable")
    return frame
