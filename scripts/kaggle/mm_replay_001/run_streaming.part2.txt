    token: str,
    bbo_state: dict[str, tuple[float, float, int, str]],
    complement: dict[str, str],
) -> tuple[float | None, int | None]:
    other = complement[token]
    state = bbo_state.get(other)
    if state is None:
        return None, None
    bid, ask, ts_ns, _ = state
    return 1.0 - (bid + ask) / 2.0, ts_ns


def emit_state_row(
    *,
    sink: CanonicalTokenSink,
    token: str,
    timestamp_ns: int,
    order_seq: int,
    source_segment: int,
    row_kind: str,
    bbo_state: dict[str, tuple[float, float, int, str]],
    execution: dict[str, dict[str, Any]],
    complement: dict[str, str],
    window_id: str,
) -> bool:
    local = bbo_state.get(token)
    if local is None:
        return False
    bid, ask, _, source_version = local
    ext, ext_ts = external_fields(token, bbo_state, complement)
    meta = execution[token]
    sink.add(
        token,
        {
            "token_id": token,
            "timestamp_ns": int(timestamp_ns),
            "order_seq": int(order_seq),
            "source_segment": int(source_segment),
            "row_kind": row_kind,
            "best_bid": bid,
            "best_ask": ask,
            "external_fv": ext,
            "external_fv_timestamp_ns": ext_ts,
            "trade_price": None,
            "trade_size": None,
            "aggressor_side": None,
            "mapping_class": meta["mapping_class"],
            "sig_market_id": meta["sig_market_id"],
            "sig_exchange_id": meta["sig_exchange_id"],
            "window_id": window_id,
            "source_version": source_version,
        },
    )
    return True


def emit_trade_row(
    *,
    sink: CanonicalTokenSink,
    token: str,
    timestamp_ns: int,
    order_seq: int,
    source_segment: int,
    trade_price: Any,
    trade_size: Any,
    trade_side: Any,
    source_version: str,
    bbo_state: dict[str, tuple[float, float, int, str]],
    execution: dict[str, dict[str, Any]],
    complement: dict[str, str],
    window_id: str,
) -> bool:
    local = bbo_state.get(token)
    if local is None:
        return False
    try:
        price = float(trade_price)
        size = float(trade_size)
    except (TypeError, ValueError):
        return False
    side = str(trade_side or "").upper()
    if not (0.0 < price < 1.0 and size > 0.0 and math.isfinite(size) and side in {"BUY", "SELL"}):
        return False
    bid, ask, _, local_source = local
    ext, ext_ts = external_fields(token, bbo_state, complement)
    meta = execution[token]
    sink.add(
        token,
        {
            "token_id": token,
            "timestamp_ns": int(timestamp_ns),
            "order_seq": int(order_seq),
            "source_segment": int(source_segment),
            "row_kind": "TRADE",
            "best_bid": bid,
            "best_ask": ask,
            "external_fv": ext,
            "external_fv_timestamp_ns": ext_ts,
            "trade_price": price,
            "trade_size": size,
            "aggressor_side": side,
            "mapping_class": meta["mapping_class"],
            "sig_market_id": meta["sig_market_id"],
            "sig_exchange_id": meta["sig_exchange_id"],
            "window_id": window_id,
            "source_version": source_version or local_source,
        },
    )
    return True


def canonicalize_window(
    *,
    window_id: str,
    files: list[Path],
    execution: dict[str, dict[str, Any]],
    complement: dict[str, str],
) -> tuple[Path, dict[str, Any]]:
    out_root = TMP / window_id
    if out_root.exists():
        shutil.rmtree(out_root)
    sink = CanonicalTokenSink(out_root)
    required_tokens = set(execution) | set(complement.values())
    complement_to_exec = {other: token for token, other in complement.items()}
    bbo_state: dict[str, tuple[float, float, int, str]] = {}
    last_hour: pd.Timestamp | None = None
    segment = 0
    order_seq = 0
    stats: Counter[str] = Counter()

    for file_index, path in enumerate(sorted(files, key=hour_from_path)):
        hour = hour_from_path(path)
        if last_hour is not None and hour - last_hour > pd.Timedelta(hours=1, minutes=1):
            segment += 1
            bbo_state.clear()
            stats["source_gap_segments"] += 1
        last_hour = hour
        frame = filtered_flat_frame(path, required_tokens)
        stats["files"] += 1
        if frame.empty:
            continue
        stats["filtered_rows"] += len(frame)
        for timestamp_ns, group in frame.groupby("__ts_ns", sort=True, observed=True):
            # Fill evidence is evaluated before same-timestamp BBO changes, so a
            # quote never benefits from a contemporaneous book update whose
            # ordering is not guaranteed by the research timestamp.
            trades = group[group["event_type"] == "last_trade_price"]
            for rec in trades.to_dict(orient="records"):
                token = str(rec["__token"])
                if token not in execution:
                    continue
                order_seq += 1
                if emit_trade_row(
                    sink=sink,
                    token=token,
                    timestamp_ns=int(timestamp_ns),
                    order_seq=order_seq,
                    source_segment=segment,
                    trade_price=rec.get("price"),
                    trade_size=rec.get("size"),
                    trade_side=rec.get("side"),
                    source_version=str(rec.get("source_version") or "PMXT"),
                    bbo_state=bbo_state,
                    execution=execution,
                    complement=complement,
                    window_id=window_id,
                ):
                    stats["trades_emitted"] += 1
                else:
                    stats["trades_skipped_no_state_or_invalid"] += 1

            changed: set[str] = set()
            ambiguous: set[str] = set()
            bbo_rows = group[group["event_type"].isin(["price_change", "best_bid_ask"])]
            for token, sub in bbo_rows.groupby("__token", sort=False, observed=True):
                values: set[tuple[float, float]] = set()
                source_version = "PMXT"
                for rec in sub.to_dict(orient="records"):
                    parsed = valid_bbo(rec.get("best_bid"), rec.get("best_ask"))
                    if parsed is not None:
                        values.add(parsed)
                        source_version = str(rec.get("source_version") or source_version)
                if len(values) > 1:
                    ambiguous.add(str(token))
                    stats["ambiguous_same_timestamp_bbo"] += 1
                    continue
                if len(values) != 1:
                    continue
                value = next(iter(values))
                old = bbo_state.get(str(token))
                if old is not None and math.isclose(old[0], value[0], abs_tol=1e-12, rel_tol=0.0) and math.isclose(old[1], value[1], abs_tol=1e-12, rel_tol=0.0):
                    stats["repeated_bbo_groups"] += 1
                    continue
                bbo_state[str(token)] = (value[0], value[1], int(timestamp_ns), source_version)
                changed.add(str(token))
                stats["genuine_bbo_groups"] += 1

            impacted: dict[str, str] = {}
            for changed_token in changed:
                if changed_token in execution:
                    impacted[changed_token] = "LOCAL_BBO"
                external_target = complement_to_exec.get(changed_token)
                if external_target is not None and external_target not in impacted:
                    impacted[external_target] = "EXTERNAL_BBO"
            for token, kind in impacted.items():
                order_seq += 1
                if emit_state_row(
                    sink=sink,
                    token=token,
                    timestamp_ns=int(timestamp_ns),
                    order_seq=order_seq,
                    source_segment=segment,
                    row_kind=kind,
                    bbo_state=bbo_state,
                    execution=execution,
                    complement=complement,
                    window_id=window_id,
                ):
                    stats[f"{kind.lower()}_emitted"] += 1

        if file_index % 24 == 0:
            print(
                f"[{window_id}] {file_index + 1}/{len(files)} files; "
                f"filtered={stats['filtered_rows']} emitted={sum(sink.counts.values())}",
                flush=True,
            )
    sink.close()
    stats.update({f"canonical_{k.lower()}": v for k, v in sink.counts.items()})
    stats["token_files"] = len(sink.paths)
    stats["segments"] = segment + 1 if files else 0
    return out_root, dict(stats)


def row_to_observation(rec: dict[str, Any], hazard: float | None = None) -> BookObservation:
    side_text = str(rec.get("aggressor_side") or "").upper()
    side = Side(side_text) if side_text in {"BUY", "SELL"} else None
    ext = rec.get("external_fv")
    ext_ts = rec.get("external_fv_timestamp_ns")
    trade_price = rec.get("trade_price")
    trade_size = rec.get("trade_size")
    return BookObservation(
        market_id=str(rec["token_id"]),
        timestamp_ns=int(rec["timestamp_ns"]),
        best_bid=float(rec["best_bid"]) if pd.notna(rec.get("best_bid")) else None,
        best_ask=float(rec["best_ask"]) if pd.notna(rec.get("best_ask")) else None,
        external_fv=float(ext) if pd.notna(ext) else None,
        external_fv_timestamp_ns=int(ext_ts) if pd.notna(ext_ts) else None,
        trade_price=float(trade_price) if pd.notna(trade_price) else None,
        trade_size=float(trade_size) if pd.notna(trade_size) else None,
        aggressor_side=side,
        update_hazard=hazard,
        category=str(rec.get("mapping_class") or ""),
    )


def dual_005f_scores(
    local_bbo_observations: list[BookObservation],
    *,
    grid_origin_ns: int,
    artifacts: dict[str, tuple[Any, Any, dict[str, Any]]],
    transfer_sink: CsvSink,
    transfer_stats: dict[str, Any],
    window_id: str,
    source_segment: int,
    mapping_class: str,
    sig_market_id: str,
) -> list[tuple[int, float]]:
    pre_key = "PRE_ELECTION|clock|UPDATE_HAZARD"
    active_key = "ACTIVE_RESULTS|clock|UPDATE_HAZARD"
    if pre_key not in artifacts or active_key not in artifacts or not local_bbo_observations:
        return []
    grouped = group_bbo_for_005f(local_bbo_observations)
    if not grouped:
        return []
    adapter = Frozen005FTransferAdapter(
        scope_id=local_bbo_observations[0].market_id,
        grid_origin_ns=grid_origin_ns,
    )
    for obs in grouped:
        adapter.observe(
            timestamp_ns=obs.timestamp_ns,
            best_bid=obs.best_bid,
            best_ask=obs.best_ask,
            ambiguous=obs.ambiguous,
        )
    genuine = list(genuine_005f_change_times(grouped))
    first = local_bbo_observations[0].timestamp_ns
    last = local_bbo_observations[-1].timestamp_ns
    query = grid_origin_ns
    if query < first:
        query += math.ceil((first - query) / GRID_NS) * GRID_NS
    scores: list[tuple[int, float]] = []
    pre_model, pre_scaler, pre_row = artifacts[pre_key]
    act_model, act_scaler, act_row = artifacts[active_key]
    while query + 300_000_000_000 <= last:
        features_raw = adapter.features(query_timestamp_ns=query)
        if features_raw is not None:
            features = dict(features_raw)
            pre = legacy.score_artifact(pre_model, pre_scaler, pre_row, features)
            active = legacy.score_artifact(act_model, act_scaler, act_row, features)
            risk = max(pre, active) if math.isfinite(pre) and math.isfinite(active) else math.nan
            a = bisect_right(genuine, query)
            b = bisect_right(genuine, query + 300_000_000_000)
            target = float(b > a)
            transfer_sink.write(
                [
                    {
                        "market_id": local_bbo_observations[0].market_id,
                        "window_id": window_id,
                        "source_segment": source_segment,
                        "mapping_class": mapping_class,
                        "sig_market_id": sig_market_id,
                        "grid_time_ns": query,
                        "update_h300": target,
                        "genuine_15": features.get("genuine_15"),
                        "genuine_60": features.get("genuine_60"),
                        "genuine_age_s": features.get("genuine_age_s"),
                        "pre_frozen_score": pre,
                        "active_frozen_score": active,
                        "risk_max_score": risk,
                    }
                ]
            )
            if math.isfinite(pre) and math.isfinite(active) and math.isfinite(risk):
                transfer_stats["rows"] += 1
                transfer_stats["target_sum"] += target
                transfer_stats["brier_pre_sum"] += (pre - target) ** 2
                transfer_stats["brier_active_sum"] += (active - target) ** 2
                transfer_stats["brier_risk_max_sum"] += (risk - target) ** 2
                bin_id = min(9, max(0, int(risk * 10.0)))
                transfer_stats["risk_bins_count"][bin_id] += 1
                transfer_stats["risk_bins_target_sum"][bin_id] += target
                transfer_stats["risk_bins_score_sum"][bin_id] += risk
                scores.append((query, risk))
        query += GRID_NS
    return scores


def attach_hazard(
    frame: pd.DataFrame,
    scores: list[tuple[int, float]],
) -> list[BookObservation]:
    if not scores:
        return [row_to_observation(rec) for rec in frame.to_dict(orient="records")]
    times = [item[0] for item in scores]
    observations: list[BookObservation] = []
    for rec in frame.to_dict(orient="records"):
        ts = int(rec["timestamp_ns"])
        index = bisect_right(times, ts) - 1
        hazard = scores[index][1] if index >= 0 else None
        observations.append(row_to_observation(rec, hazard))
    return observations


def split_for_window(timestamp_ns: int, start_ns: int, end_ns: int) -> str:
    span = max(1, end_ns - start_ns)
    fraction = (timestamp_ns - start_ns) / span
    if fraction < 0.60:
        return "TRAIN"
    if fraction < 0.80:
        return "DEV"
    return "FINAL"
