def quote_series(corpus:Path,event:str,tokens:list[str],start:datetime,end:datetime)->dict[str,dict[str,np.ndarray]]:

def project_simplex(v:np.ndarray)->np.ndarray:
 x=np.asarray(v,float);u=np.sort(x)[::-1];cssv=np.cumsum(u)-1;rho=np.nonzero(u-cssv/(np.arange(len(u))+1)>0)[0]
 if len(rho)==0:return np.full_like(x,1/len(x))
 theta=cssv[rho[-1]]/(rho[-1]+1);return np.clip(x-theta,0,1)
def project_capped(v:np.ndarray)->np.ndarray:
 x=np.clip(np.asarray(v,float),0,1);return x if float(np.sum(x))<=1+1e-12 else project_simplex(x)
def project_nonincreasing(v:np.ndarray)->np.ndarray:
 y=-np.asarray(v,float);means=[];weights=[]
 for val in y:
  means.append(float(val));weights.append(1)
  while len(means)>=2 and means[-2]>means[-1]:
   w=weights[-2]+weights[-1];mean=(means[-2]*weights[-2]+means[-1]*weights[-1])/w
   means[-2:]=[mean];weights[-2:]=[w]
 out=np.empty(len(y),float);pos=0
 for mean,w in zip(means,weights):out[pos:pos+w]=-mean;pos+=w
 return np.clip(out,0,1)
def residual_matrix(values:np.ndarray,relation:str)->np.ndarray:
 out=np.full_like(values,np.nan);finite=np.all(np.isfinite(values),axis=1)
 if relation=='MUTUALLY_EXCLUSIVE_NONEXHAUSTIVE':
  x=np.clip(values[finite],0,1);res=np.zeros_like(x);viol=np.sum(x,axis=1)>1+1e-12
  if np.any(viol):
   v=x[viol];u=np.sort(v,axis=1)[:,::-1];cssv=np.cumsum(u,axis=1)-1
   k=np.arange(1,v.shape[1]+1);rho=np.sum(u-cssv/k>0,axis=1)-1
   theta=cssv[np.arange(len(v)),rho]/(rho+1);proj=np.maximum(v-theta[:,None],0)
   res[viol]=v-proj
  out[finite]=res;return out
 for i,row in enumerate(values):
  if not finite[i]:continue
  if relation=='EXHAUSTIVE_PARTITION':p=project_simplex(row)
  elif relation=='CONDITIONAL_STAGED':p=project_nonincreasing(row)
  else:raise ValueError(relation)
  out[i]=row-p
 return out

def controlled_beta(rows:dict[str,np.ndarray],signal:np.ndarray)->tuple[float|None,int]:
 t=rows['eval_t'];j=rows['eval_j'];y=rows['eval_y'];c=rows['eval_c']
 if len(t)<MIN_OVERLAP:return None,len(t)
 x=signal[t,j];mask=np.isfinite(x)&np.isfinite(y)&np.all(np.isfinite(c),axis=1);n=int(mask.sum())
 if n<MIN_OVERLAP:return None,n
 x=x[mask];y=y[mask];c=c[mask];C=np.column_stack([np.ones(n),c])
 yr=y-C@np.linalg.lstsq(C,y,rcond=None)[0];xr=x-C@np.linalg.lstsq(C,x,rcond=None)[0];den=float(xr@xr)
 return (None,n) if den<=1e-18 else (float((xr@yr)/den),n)
def raw_beta(rows:dict[str,np.ndarray],signal:np.ndarray)->float|None:
 if len(rows['eval_t'])<MIN_OVERLAP:return None
 x=signal[rows['eval_t'],rows['eval_j']];y=rows['eval_y'];m=np.isfinite(x)&np.isfinite(y)
 den=float(x[m]@x[m]) if np.any(m) else 0
 return None if int(m.sum())<MIN_OVERLAP or den<=1e-18 else float((x[m]@y[m])/den)
def alignment(rows:dict[str,np.ndarray],signal:np.ndarray)->float|None:

 x=signal[rows['eval_t'],rows['eval_j']];y=rows['eval_y'];m=np.isfinite(x)&np.isfinite(y)

def soft_signal(panel:dict[str,Any],idx:list[int])->np.ndarray:

def shift_signal(x:np.ndarray,rng:np.random.Generator,kind:str)->np.ndarray:
 T=len(x)
 if kind=='CIRCULAR_SHIFT':
  lo=min(20,max(1,T//4));hi=max(lo+1,T-lo);k=int(rng.integers(lo,hi));return np.roll(x,k,axis=0)

 order=rng.permutation(len(blocks));return np.concatenate([blocks[k] for k in order],axis=0)[:T]
def empirical_p(obs:float|None,draws:list[float])->float|None:

  summary[kind]={'valid_draws':len(draws),'p_value':empirical_p(obs,draws),'mean':float(np.mean(draws)) if draws else None,'std':float(np.std(draws)) if draws else None}
