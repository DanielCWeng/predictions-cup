def corr(x:np.ndarray,y:np.ndarray)->tuple[int,float|None]:
 mask=np.isfinite(x)&np.isfinite(y); n=int(mask.sum())
 if n<MIN_OVERLAP:return n,None
 a=x[mask]; b=y[mask]
 if np.std(a)<=1e-15 or np.std(b)<=1e-15:return n,None
 return n,float(np.corrcoef(a,b)[0,1])
def slope_1sd(x:np.ndarray,y:np.ndarray)->tuple[int,float|None,float|None,float|None]:
 mask=np.isfinite(x)&np.isfinite(y); n=int(mask.sum())
 if n<MIN_OVERLAP:return n,None,None,None
 a=x[mask]; b=y[mask]; scale=float(np.std(a))
 if scale<=1e-15 or np.std(b)<=1e-15:return n,None,None,scale
 z=(a-np.mean(a))/scale; yc=b-np.mean(b); denom=float(np.dot(z,z))
 if denom<=0:return n,None,None,scale
 effect=float(np.dot(z,yc)/denom); c=float(np.corrcoef(z,b)[0,1]); return n,effect,c,scale

def pairwise_corr_matrix(X:np.ndarray,Y:np.ndarray|None=None)->tuple[np.ndarray,np.ndarray]:
 if Y is None:Y=X
 mx=np.isfinite(X).astype(float); my=np.isfinite(Y).astype(float); a=np.nan_to_num(X); b=np.nan_to_num(Y)
 n=mx.T@my; sx=a.T@my; sy=mx.T@b; sxx=(a*a).T@my; syy=mx.T@(b*b); sxy=a.T@b
 with np.errstate(divide='ignore',invalid='ignore'):
  cov=sxy-sx*sy/n; vx=sxx-sx*sx/n; vy=syy-sy*sy/n; c=cov/np.sqrt(vx*vy)
 c[(n<MIN_OVERLAP)|(vx<=1e-30)|(vy<=1e-30)]=np.nan
 return c,n

def circular(X:np.ndarray,rng:np.random.Generator,min_bins:int)->np.ndarray:
 T,M=X.shape; out=np.empty_like(X)
 for j in range(M):
  if T<=2*min_bins: shift=max(1,T//2)
  else: shift=int(rng.integers(min_bins,T-min_bins+1))
  out[:,j]=np.roll(X[:,j],shift)
 return out
def block_perm(X:np.ndarray,rng:np.random.Generator,block:int)->np.ndarray:
 T,M=X.shape; out=np.empty_like(X)
 blocks=[(s,min(T,s+block)) for s in range(0,T,block)]

  order=rng.permutation(len(blocks)); vals=np.concatenate([X[s:e,j] for s,e in (blocks[k] for k in order)])
  out[:,j]=vals[:T]
 return out

def ranks(v:np.ndarray)->np.ndarray:
 order=np.argsort(v,kind='mergesort'); r=np.empty(len(v),float); i=0
 while i<len(v):
  j=i+1
  while j<len(v) and v[order[j]]==v[order[i]]:j+=1
  r[order[i:j]]=(i+j-1)/2+1; i=j
 return r
