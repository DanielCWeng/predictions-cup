from __future__ import annotations

import hashlib
import json
import subprocess
import sys
from pathlib import Path

INPUT = Path("/kaggle/input")
WORK = Path("/kaggle/working")
EXPECTED_COMMIT = "b2ba5e2c3e63d57bd6e74c0c5861bad67c983e1b"
EXPECTED_CORPUS = "e3d95735262b0be3e9fe7dd32fb54e53dc7e65b403ef9b5a2261ac107a8ea4c4"
EXPECTED_PREREG = "64af4c1427afa983a13da6a11d5721b1b6496b0306ce1b07ab6e2f19df9e7e70"
EXPECTED_RELATIONSHIPS = "e93033568ac3d3045399e6b630d37bb30a829c9c2fad3ef5da8ac31a7516732e"


def sha(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def one(name: str) -> Path:
    matches = sorted(INPUT.rglob(name))
    if len(matches) != 1:
        raise SystemExit(f"expected one {name}, got {matches}")
    return matches[0]


wheel = one("predictions_cup-0.1.0-py3-none-any.whl")
commit = one("COMMIT.txt")
prereg = one("preregistration.json")
relationships = one("relationship_inventory.csv")
manifest = one("corpus_manifest.json")
if commit.read_text().strip() != EXPECTED_COMMIT:
    raise SystemExit("code commit marker mismatch")
if sha(prereg) != EXPECTED_PREREG:
    raise SystemExit("preregistration hash mismatch")
if sha(relationships) != EXPECTED_RELATIONSHIPS:
    raise SystemExit("relationship inventory hash mismatch")
if sha(manifest) != EXPECTED_CORPUS:
    raise SystemExit("corpus manifest hash mismatch")
corpus_root = manifest.parent.parent

subprocess.check_call([sys.executable, "-m", "pip", "install", "-q", "--no-deps", str(wheel)])
import numpy
import pyarrow
print(json.dumps({
    "phase": "environment_verified",
    "code_commit": EXPECTED_COMMIT,
    "corpus_manifest_sha256": sha(manifest),
    "preregistration_sha256": sha(prereg),
    "relationship_inventory_sha256": sha(relationships),
    "numpy": numpy.__version__,
    "pyarrow": pyarrow.__version__,
    "corpus_root": str(corpus_root),
}, sort_keys=True))

out = WORK / "experiment_003"
cmd = [
    sys.executable, "-m", "predictions_cup.learning.historical_alpha",
    "--corpus-root", str(corpus_root),
    "--preregistration", str(prereg),
    "--relationship-inventory", str(relationships),
    "--output-dir", str(out),
    "--code-revision", EXPECTED_COMMIT,
]
print(json.dumps({"phase": "launch", "command": cmd}, sort_keys=True))
subprocess.check_call(cmd)
summary = json.loads((out / "battery_summary.json").read_text())
print(json.dumps({
    "phase": "complete",
    "run_id": summary["run_id"],
    "families": summary["families"],
}, sort_keys=True))
